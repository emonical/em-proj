def lock_acquire(name: str, ttl: int = DEFAULT_TTL, reason: str | None = None) -> dict:
    """Acquire state:lock:<name> for the current process composite.
    - validate_key(name) per Phase 2 D-09 carry
    - validate reason (max length 256, D-12 discretion)
    - SET NX EX with the holder JSON
    - on collision: read existing holder, run is_holder_stale; if stale, attempt Lua compare-and-swap
    - on still-held: block-poll up to DEFAULT_BLOCK_SECONDS, then raise HeldByAnother(holder=...)

    - validate_key(name)
    - If the lock has been displaced (compare-and-delete script returns "not_holder"), raise HeldByAnother(holder=displacer_dict_or_None) per D-09 (displaced holder learns).
    - If the lock is already gone (expired/forced-unlocked), raise HeldByAnother(holder=None).
    """

def lock_force_displace(name: str, ttl: int = DEFAULT_TTL, reason: str | None = None) -> dict:
    """Atomically replace any current holder of state:lock:<name> with a fresh holder
    JSON for the calling process. Used by --warn override (Plan 03-04) to keep the
    displacement Lua server-side and the verb body thin (D-14 / D-17).
